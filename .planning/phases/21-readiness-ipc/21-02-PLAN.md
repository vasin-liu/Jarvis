        let model = config.embedding.fastembed_model.clone();

        std::thread::spawn(move || match FastEmbedder::try_new(&model, cache) {

            }

        (embedder, deferred_embedder)

    #[serde(rename_all = "camelCase")]
    pub struct EmbedderReadinessView {
        pub state: String,
        pub message: Option<String>,
    }

    fn readiness_view(state: EmbedderReadyState) -> EmbedderReadinessView {
        match state {
            EmbedderReadyState::Pending => EmbedderReadinessView {
                state: "pending".into(),
                message: None,

            EmbedderReadyState::Ready => EmbedderReadinessView {
                state: "ready".into(),
                message: None,

                state: "failed".into(),

        }
